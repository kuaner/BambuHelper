cmake_minimum_required(VERSION 3.20)
project(p4_cloud_camera_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenSSL REQUIRED)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(TRANSPORT "${ROOT}/lib/BambuTUTK/src")
set(ARDUINOJSON "${ROOT}/.pio/libdeps/ws_p4_lcd_7b_rev1/ArduinoJson/src" CACHE PATH "ArduinoJson source directory")
add_library(camera_transport STATIC ${TRANSPORT}/IotcClient.cpp ${TRANSPORT}/TutkSession.cpp ${TRANSPORT}/Fec.cpp ${TRANSPORT}/Snapshot.cpp)
target_include_directories(camera_transport PUBLIC ${TRANSPORT} ${ARDUINOJSON})
option(BAMBU_CAMERA_MBEDTLS "Exercise the P4 crypto backend on the host" OFF)
if(BAMBU_CAMERA_MBEDTLS)
  set(MBEDTLS_SOURCE "${ROOT}/.cache/reference/mbedtls-3.6.6" CACHE PATH "Mbed TLS 3.6.6 source")
  set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  add_subdirectory(${MBEDTLS_SOURCE} ${CMAKE_BINARY_DIR}/mbedtls EXCLUDE_FROM_ALL)
  target_compile_definitions(camera_transport PUBLIC BAMBU_CAMERA_MBEDTLS=1)
  target_link_libraries(camera_transport PUBLIC mbedcrypto)
else()
  target_link_libraries(camera_transport PUBLIC OpenSSL::Crypto)
endif()
target_compile_options(camera_transport PRIVATE -Wall -Wextra -Wno-deprecated-declarations)
option(BAMBU_CAMERA_DEBUG "Enable redacted transport diagnostics" OFF)
if(BAMBU_CAMERA_DEBUG)
  target_compile_definitions(camera_transport PUBLIC BAMBU_CAMERA_DEBUG=1)
endif()
add_executable(camera_probe main.cpp)
target_link_libraries(camera_probe PRIVATE camera_transport)
enable_testing()
add_executable(snapshot_tests tests.cpp)
target_link_libraries(snapshot_tests PRIVATE camera_transport OpenSSL::Crypto)
add_test(NAME snapshot_protocol COMMAND snapshot_tests)
